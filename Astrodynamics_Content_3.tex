\begin{frame}{Continuous Thrust Trajectories: The Paradigm Shift}
    \begin{columns}[T]
        \begin{column}{0.49\textwidth}
            \begin{block}{The Classical Impulsive Model}
                \scriptsize
                \begin{itemize}\setlength{\itemsep}{2pt}\setlength{\topsep}{1pt}
                    \item \textbf{Infinitesimal Burn Duration:} Assumes burn time $\Delta t \to 0$.
                    \item \textbf{Invariant Position:} Spacecraft position $\mathbf{r}$ remains frozen during velocity change.
                    \item \textbf{Instantaneous Vector Jump:} $\mathbf{v}^+ = \mathbf{v}^- + \Delta \mathbf{v}$.
                    \item Localized to optimal apsidal points (e.g., periapsis Oberth burns).
                \end{itemize}
            \end{block}
            \vspace{-0.1cm}
            \begin{alertblock}{The Operational Reality: Finite Maneuvers}
                \scriptsize
                \begin{itemize}\setlength{\itemsep}{2pt}\setlength{\topsep}{1pt}
                    \item Chemical engines burn for minutes; Solar Electric Propulsion (SEP) thrusters burn for weeks or months.
                    \item Spacecraft position $\mathbf{r}(t)$ changes significantly during the burn arc.
                    \item Acceleration is sub-Newtonian ($10^{-3}$ to $10^{-6}$ g), requiring continuous integration over many revolutions.
                \end{itemize}
            \end{alertblock}
        \end{column}

        \begin{column}{0.49\textwidth}
            \begin{block}{The Fundamental Trade-Off: Mass vs. Time}
                \scriptsize
                \begin{itemize}\setlength{\itemsep}{2pt}\setlength{\topsep}{1pt}
                    \item \textbf{Gravity Loss Penalty:} Thrusting away from apsides requires higher cumulative velocity increment ($\Delta V_{\text{spiral}} > \Delta V_{\text{Hohmann}}$).
                    \item \textbf{Specific Impulse Leverage:} Electric thrusters achieve $I_{\text{sp}} \sim 3{,}000\text{--}5{,}000\text{ s}$ ($\sim 10\times$ chemical rockets).
                    \item \textbf{Tsiolkovsky Exponential Advantage:}
                    $$\frac{m_f}{m_0} = \exp\left(-\frac{\Delta V}{I_{\text{sp}} g_0}\right)$$
                    \item Drastically reduces propellant mass fraction, liberating mass for science payload.
                \end{itemize}
            \end{block}

% SLIDE 3: DYNAMIC FORMULATION: CONIC JUMPS VS 7D EOM
% ==============================================================================
\begin{frame}{Dynamic Formulation: Conic Jumps vs. 7D State Integration}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Impulsive Formulation ($\mathbb{R}^6$ Conic Jumps)}
                \scriptsize
                Maneuvers are modeled as point velocity discontinuities:
                \par\vspace{1pt}
                {\centering $\mathbf{r}^+ = \mathbf{r}^-, \quad \mathbf{v}^+ = \mathbf{v}^- + \Delta \mathbf{v}$ \par}
                \begin{itemize}\setlength{\itemsep}{1.2pt}\setlength{\topsep}{0pt}
                    \item \textbf{Position Invariance:} $\Delta t \to 0$ assumes orbital position is frozen during firing.
                    \item \textbf{Decoupled Mass Step:} Mass is subtracted discretely via the Tsiolkovsky equation after each burn:
                    \par\vspace{1pt}
                    {\centering $\Delta m = m_0 \left[1 - \exp\left(-\frac{\Delta v}{I_{\text{sp}} g_0}\right)\right]$ \par}
                    \item Propagator takes massive steps during unpowered coast arcs.
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{0.48\textwidth}
            \begin{exampleblock}{Continuous Low-Thrust Dynamics ($\mathbb{R}^7$ EOM)}
                \scriptsize
                Position changes significantly while thrusting; mass becomes a dynamic coordinate:
                \par\vspace{1pt}
                {\centering $\mathbf{Y}(t) = [\mathbf{r}^T, \; \mathbf{v}^T, \; m]^T \in \mathbb{R}^7$ \par}
                \vspace{-2pt}
                \begin{equation*}
                    \frac{d\mathbf{Y}}{dt} = \begin{bmatrix} 
                        \mathbf{v} \\ 
                        -\frac{\mu}{r^3}\mathbf{r} + \frac{\mathbf{T}(t)}{m(t)} + \mathbf{a}_{\text{pert}} \\ 
                        -\frac{\|\mathbf{T}(t)\|}{I_{\text{sp}} g_0} 
                    \end{bmatrix}
                \end{equation*}
                \begin{itemize}\setlength{\itemsep}{1.2pt}\setlength{\topsep}{0pt}
                    \item \textbf{Acceleration Coupling:} Constant $T$ causes acceleration $a(t) = T/m(t)$ to rise continuously.
                    \item \textbf{Adaptive Step Reduction:} Numerical engines must shrink step sizes across thousands of revolutions.
                \end{itemize}
            \end{exampleblock}
        \end{column}
    \end{columns}
\end{frame}

% SLIDE 4: TOOLCHAIN DIVERGENCE: TARGETERS VS OPTIMIZERS
% ==============================================================================
\begin{frame}{Flight Mechanics Toolchain: Targeters vs. Optimizers}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Impulsive Mission Workflow}
                \scriptsize
                \begin{itemize}\setlength{\itemsep}{1.5pt}\setlength{\topsep}{0pt}
                    \item \textbf{Boundary Value Engine:} Lambert's problem connects departure and arrival positions over fixed transfer times.
                    \item \textbf{Porkchop Map Exploration:} Fast scans over launch and arrival windows calculate $C_3$ and $\Delta V$ contours.
                    \item \textbf{Differential Correctors:} Local Newton-Raphson shooting methods iteratively converge on target B-plane coordinates.
                \end{itemize}
            \end{block}
            \vspace{-0.15cm}
            \begin{alertblock}{Why Simple Targeters Fail in Low-Thrust}
                \scriptsize
                \begin{itemize}\setlength{\itemsep}{1.2pt}\setlength{\topsep}{0pt}
                    \item Continuous burns span hundreds to thousands of orbits.
                    \item Sensitivities become highly cyclical and non-linear, causing differential correctors to easily diverge or encounter singular Jacobians.
                \end{itemize}
            \end{alertblock}
        \end{column}

        \begin{column}{0.48\textwidth}
            \begin{exampleblock}{Low-Thrust Mission Workflow}
                \scriptsize
                \begin{itemize}\setlength{\itemsep}{1.5pt}\setlength{\topsep}{0pt}
                    \item \textbf{Continuous Steering Laws:} Solves for a continuous thrust orientation vector history $\hat{\mathbf{u}}(t)$.
                    \item \textbf{Pontryagin's Minimum Principle (PMP):} Formulates the Hamiltonian $\mathcal{H}$ with costate tracking:
                    \par\vspace{1pt}
                    {\centering $\hat{\mathbf{u}}^*(t) = -\frac{\boldsymbol{\lambda}_v}{\|\boldsymbol{\lambda}_v\|} \quad (\text{Primer Vector Steering})$ \par}
                    \item \textbf{Nonlinear Programming (NLP):} Deploys direct collocation or optimizers (e.g., SNOPT, IPOPT) to minimize propellant under path constraints.
                \end{itemize}
            \end{exampleblock}
        \end{column}
    \end{columns}
\end{frame}

% ==============================================================================
% SLIDE 5A: PONTRYAGIN'S MINIMUM PRINCIPLE
% ==============================================================================
\begin{frame}{Optimal Control: Why Low-Thrust Needs Pontryagin (PMP)}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{From 3 Numbers to an Infinite Function}
                \scriptsize
                \begin{itemize}\setlength{\itemsep}{1.5pt}\setlength{\topsep}{0pt}
                    \item \textbf{Impulsive Burn (Static):} Optimizer only solves for 3 discrete numbers $(\Delta v_x, \Delta v_y, \Delta v_z)$ at a fixed instant $t_0$.
                    \item \textbf{Low-Thrust Burn (Continuous):} Engine fires for months. We must find a continuous 3D unit steering vector function $\hat{\mathbf{u}}(t)$ at \textit{every instant} $t \in [t_0, t_f]$.
                    \item Searching over an infinite-dimensional space of curves requires \textbf{Optimal Control Theory}.
                \end{itemize}
            \end{block}
            \vspace{-0.15cm}
            \begin{exampleblock}{What are Costates $\boldsymbol{\lambda}(t)$?}
                \scriptsize
                Every physical state $\mathbf{Y} = [\mathbf{r}, \mathbf{v}, m]^T$ is paired with a shadow \textbf{costate vector} $\boldsymbol{\lambda} = [\boldsymbol{\lambda}_r, \boldsymbol{\lambda}_v, \lambda_m]^T$ that acts as a \textit{sensitivity compass}, measuring how state changes today impact final fuel cost.
            \end{exampleblock}
        \end{column}

        \begin{column}{0.49\textwidth}
            \begin{block}{The Control Hamiltonian ($\mathcal{H}$)}
                \scriptsize
                Pontryagin combined the instantaneous cost $L$ and the 7D equations of motion ($\dot{\mathbf{Y}}$) into a single scalar scoreboard called the \textbf{Hamiltonian}:
                \par\vspace{2pt}
                {\centering $\displaystyle \mathcal{H} = L + \boldsymbol{\lambda}_r \cdot \mathbf{v} + \boldsymbol{\lambda}_v \cdot \left(\mathbf{a}_{\text{grav}} + \frac{T}{m}\hat{\mathbf{u}}(t)\right) - \lambda_m \frac{T}{I_{\text{sp}} g_0}$ \par}
                \vspace{2pt}
                \begin{itemize}\setlength{\itemsep}{1.2pt}\setlength{\topsep}{0pt}
                    \item $\boldsymbol{\lambda}_r \in \mathbb{R}^3$: Position sensitivity costate.
                    \item $\boldsymbol{\lambda}_v \in \mathbb{R}^3$: Velocity sensitivity costate.
                    \item $\lambda_m \in \mathbb{R}^1$: Mass sensitivity costate.
                \end{itemize}
            \end{block}
            \vspace{-0.15cm}
            \begin{alertblock}{Pontryagin's Minimum Principle (PMP)}
                \scriptsize
                To minimize total propellant over the entire multi-month mission, the optimal thrust direction $\hat{\mathbf{u}}^*(t)$ must minimize $\mathcal{H}$ \textbf{at every single instant $t$}:
                \par\vspace{1pt}
                {\centering $\displaystyle \hat{\mathbf{u}}^*(t) = \arg\min_{\|\hat{\mathbf{u}}\|=1} \mathcal{H}(\mathbf{Y}, \boldsymbol{\lambda}, \hat{\mathbf{u}}, t)$ \par}
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

% SLIDE 5B: PRIMER VECTOR STEERING
% ==============================================================================
\begin{frame}{Primer Vector Steering: Deriving the Optimal Thrust Law}
    \begin{columns}[T]
        \begin{column}{0.51\textwidth}
            \begin{block}{Minimizing the Dot Product in $\mathcal{H}$}
                \scriptsize
                \begin{enumerate}\setlength{\itemsep}{1.5pt}\setlength{\topsep}{0pt}
                    \item In the Hamiltonian $\mathcal{H}$, the thrust pointing vector $\hat{\mathbf{u}}(t)$ appears in only one term:
                    \par\vspace{1pt}
                    {\centering $\displaystyle \mathcal{H}_{\text{thrust}} = \frac{T}{m}\left(\boldsymbol{\lambda}_v \cdot \hat{\mathbf{u}}\right) = \frac{T}{m}\|\boldsymbol{\lambda}_v\| \|\hat{\mathbf{u}}\| \cos\theta$ \par}
                    \vspace{1pt}
                    \item Since $T/m > 0$ and $\|\hat{\mathbf{u}}\| = 1$, $\mathcal{H}$ is minimized when $\cos\theta = -1$ ($\theta = 180^\circ$).
                    \item Therefore, $\hat{\mathbf{u}}^*(t)$ must point \textbf{directly opposite} to the velocity costate $\boldsymbol{\lambda}_v(t)$:
                    \par\vspace{1.5pt}
                    {\centering $\displaystyle \hat{\mathbf{u}}^*(t) = -\frac{\boldsymbol{\lambda}_v(t)}{\|\boldsymbol{\lambda}_v(t)\|} = \frac{\mathbf{p}(t)}{\|\mathbf{p}(t)\|}$ \par}
                \end{enumerate}
            \end{block}
            \vspace{-0.18cm}
            \begin{exampleblock}{Lawden's Primer Vector $\mathbf{p}(t) = -\boldsymbol{\lambda}_v(t)$}
                \scriptsize
                \begin{itemize}\setlength{\itemsep}{1pt}\setlength{\topsep}{0pt}
                    \item Coined by Derek Lawden: $\mathbf{p}(t) = -\boldsymbol{\lambda}_v(t)$ dictates the exact 3D optimal thrust orientation at any second.
                    \item Costates evolve via Euler-Lagrange ODEs: $\dot{\boldsymbol{\lambda}}_r = -\frac{\partial \mathcal{H}}{\partial \mathbf{r}}$ and $\dot{\boldsymbol{\lambda}}_v = -\frac{\partial \mathcal{H}}{\partial \mathbf{v}} = -\boldsymbol{\lambda}_r$.
                \end{itemize}
            \end{exampleblock}
        \end{column}

        \begin{column}{0.47\textwidth}
            \begin{block}{Primer Vector Alignment Geometry}
                \centering
                \vspace{0.02cm}
                \begin{tikzpicture}[scale=0.84, >=Stealth]

            \end{block}
            \vspace{-0.18cm}
            \begin{alertblock}{Bang-Bang Switching Function}
                \scriptsize
                PMP also determines \textit{when to coast}: if the switching function $S(t) = \frac{\|\boldsymbol{\lambda}_v\|}{m} - \frac{\lambda_m}{I_{\text{sp}}g_0} < 0$, the thruster turns off ($T = 0$).
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

% SLIDE 5C: INDIRECT (PMP) VS DIRECT (NLP)
% ==============================================================================
\begin{frame}{Solving Low-Thrust Trajectories: Indirect PMP vs. Direct NLP}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{1. Indirect Methods (PMP + Costates)}
                \scriptsize
                \textit{``Optimize First, Then Discretize''}
                \begin{itemize}\setlength{\itemsep}{1.2pt}\setlength{\topsep}{0pt}
                    \item Combines the 7 physical states ($\mathbf{Y}$) and 7 shadow costates ($\boldsymbol{\lambda}$) into a \textbf{14D Two-Point Boundary Value Problem (TPBVP)}.
                    \item \textbf{Pros:} Mathematically exact; captures smooth analytical steering laws ($\hat{\mathbf{u}}^* = -\boldsymbol{\lambda}_v/\|\boldsymbol{\lambda}_v\|$).
                    \item \textbf{Cons:} Costates $\boldsymbol{\lambda}(t_0)$ have no intuitive physical units and a tiny radius of convergence.
                \end{itemize}
            \end{block}
            \vspace{-0.15cm}
            \begin{exampleblock}{2. Direct Methods (NLP / Collocation)}
                \scriptsize
                \textit{``Discretize First, Then Optimize''}
                \begin{itemize}\setlength{\itemsep}{1.2pt}\setlength{\topsep}{0pt}
                    \item Chops the trajectory into $N$ discrete time segments (nodes) and parameterizes thrust angles $(\alpha_k, \beta_k)$ directly.
                    \item Solved via large-scale \textbf{Nonlinear Programming (NLP)} engines (\texttt{SNOPT}, \texttt{IPOPT}, \texttt{VF13ad}).
                \end{itemize}
            \end{exampleblock}
        \end{column}

        \begin{column}{0.49\textwidth}
            \begin{block}{Architecture Comparison for GMAT and Python}
                \scriptsize
                \centering
                \renewcommand{\arraystretch}{1.15}
                \begin{tabular}{lcc}
                    \toprule
                    \textbf{Feature} & \textbf{Indirect (PMP)} & \textbf{Direct (NLP)} \\
                    \midrule
                    Control Law & Primer Vector $\mathbf{p}(t)$ & Nodal Angles $\mathbf{u}_k$ \\
                    Variables & $\mathbf{Y}(t_0), \boldsymbol{\lambda}(t_0) \in \mathbb{R}^{14}$ & hundreds of nodes \\
                    Initial Guess & Extremely sensitive & Robust / Intuitive \\
                    Path Constraints & Hard to enforce & Easy (bounds) \\
                    GMAT / Tools & Custom Python ODE & \texttt{SNOPT} / \texttt{VF13ad} \\
                    \bottomrule
                \end{tabular}
            \end{block}
            \vspace{-0.15cm}
            \begin{alertblock}{How NASA GMAT Implements This}
                \scriptsize
                In GMAT, we replace \texttt{Target} (\texttt{DifferentialCorrector}) with an \textbf{\texttt{Optimize}} sequence (\texttt{VF13ad} or \texttt{SNOPT}):
                \begin{itemize}\setlength{\itemsep}{0.5pt}\setlength{\topsep}{0pt}
                    \item \textbf{\texttt{Vary}:} Thrust pointing angles and finite burn durations.
                    \item \textbf{\texttt{NonlinearConstraint}:} Target orbital elements ($a_f, i_f, e_f$).
                    \item \textbf{\texttt{Minimize}:} Total propellant consumed ($\Delta m$) or burn time.
                \end{itemize}
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Electric Propulsion Physics: Energy Decoupling and The Power Law}
    \begin{columns}[T]
        \begin{column}{0.50\textwidth}
            \begin{block}{Decoupling Energy from Propellant Mass}
                \scriptsize
                \begin{itemize}\setlength{\itemsep}{1.5pt}\setlength{\topsep}{0pt}
                    \item \textbf{Chemical Rockets:} Energy is stored in molecular bonds. Thermodynamics caps exhaust velocity:
                    \par\vspace{1pt}
                    {\centering $v_e = \sqrt{2 c_p T_c} \le 4.5\text{ km/s} \implies I_{\text{sp}} \le 450\text{ s}$ \par}
                    \item \textbf{Electric Propulsion (EP):} Power is supplied externally (solar arrays or nuclear) and deposited into propellant particles.
                    \item Accelerates ions to hyperthermal exhaust speeds:
                    \par\vspace{1pt}
                    {\centering $v_e = g_0 I_{\text{sp}} \approx 20\text{--}50\text{ km/s} \implies I_{\text{sp}} \approx 1{,}500\text{--}5{,}000\text{ s}$ \par}
                \end{itemize}
            \end{block}
            \vspace{-0.15cm}
            \begin{alertblock}{The Fundamental Power-Thrust Dilemma}
                \scriptsize
                Kinetic beam power is $P_{\text{jet}} = \frac{1}{2} \dot{m} v_e^2 = \frac{1}{2} T g_0 I_{\text{sp}}$. For a spacecraft bus with available electrical power $P_{\text{elec}}$ and thruster efficiency $\eta$:
                \par\vspace{1pt}
                {\centering $\displaystyle T = \frac{2 \eta P_{\text{elec}}}{g_0 I_{\text{sp}}}, \qquad \dot{m} = \frac{2 \eta P_{\text{elec}}}{(g_0 I_{\text{sp}})^2} = \frac{T}{g_0 I_{\text{sp}}}$ \par}
            \end{alertblock}
        \end{column}

        \begin{column}{0.48\textwidth}
            \begin{exampleblock}{The Core Engineering Trade-Off}
                \scriptsize
                For a fixed power budget ($P_{\text{elec}} = \text{const}$):
                \begin{itemize}\setlength{\itemsep}{1.5pt}\setlength{\topsep}{0pt}
                    \item Doubling $I_{\text{sp}}$ **cuts thrust $T$ strictly in half**!
                    \item High $I_{\text{sp}}$ minimizes propellant mass, but low thrust forces burns to extend over hundreds to thousands of orbital revolutions.
                \end{itemize}
            \end{exampleblock}

LowThrustBurn.Thrusters = {HallThruster};

BeginMissionSequence;

Optimize YukonOpt {SolveMode = Solve, ExitMode = SaveAndContinue};

    % 1. VARY: Burn duration and Out-of-Plane Steering
    Vary YukonOpt(BurnDuration = 12000, ...
         {Perturbation = 1.0, Lower = 1000, Upper = 40000});
    Vary YukonOpt(HallThruster.ThrustDirection2 = 0.50, ...
         {Perturbation = 1e-4, Lower = -1.0, Upper = 1.0});

    BeginFiniteBurn LowThrustBurn(Sat);
    Propagate RK78Prop(Sat) {Sat.ElapsedSecs = BurnDuration};
    EndFiniteBurn LowThrustBurn(Sat);

    % 3. CONSTRAINT: Enforce target Semi-Major Axis
    NonlinearConstraint YukonOpt(Sat.Earth.SMA = 7000.0);

    % 4. MINIMIZE: Fuel mass expended
    FuelSpent = 50.0 - Sat.XenonTank.FuelMass;
    Minimize YukonOpt(FuelSpent);

EndOptimize;
\end{lstlisting}

                \scriptsize
                \textbf{Operating Zone:} Deep in the gravitational potential well ($r \ll r_{\text{target}}$).
                \begin{itemize}\setlength{\itemsep}{1.5pt}\setlength{\topsep}{1pt}
                    \item \textbf{Objective:} Inject orbital energy $\mathcal{E}$ as fast as possible to escape high-radiation zones.
                    \item \textbf{Kinematic Advantage:} Mechanical energy addition rate $\dot{\mathcal{E}} = \mathbf{v} \cdot \mathbf{a}_T$ is maximized when orbital speed $v$ is highest.
                    \item Firing at 100\% throttle along velocity ($\hat{\mathbf{u}} = \mathbf{v}/v$) continuously expands semi-major axis monotonically.
                \end{itemize}

            \vspace{-0.15cm}

                \scriptsize
                If 100\% thrust continues all the way to target arrival:
                \begin{itemize}\setlength{\itemsep}{1pt}\setlength{\topsep}{0pt}
                    \item \textbf{Energy Overshoot:} Spacecraft blows past $a_{\text{target}}$ into a super-synchronous graveyard orbit.
                    \item \textbf{Eccentricity Locking:} Low circular speeds at high altitude freeze orbital eccentricity errors ($e > 0$).
                \end{itemize}

                    \item \textbf{Governor Logic:} Throttle $\eta \in [0, 1]$ modulates dynamically or pulses through duty cycles:
                    \par\vspace{1pt}
                    {\centering $\displaystyle \eta(t) = \text{clip}\left(K_p \frac{a_{\text{tgt}} - a(t)}{a_{\text{tgt}}} - K_d \, e(t), \; 0, \; 1\right)$ \par}

                \begin{itemize}\setlength{\itemsep}{1.5pt}\setlength{\topsep}{0pt}

                    \par\vspace{1pt}

                    \item \textbf{Earth Umbra / Penumbra Eclipses:} In LEO, spacecraft spend $\sim 35\text{ minutes}$ of every $90\text{ minute}$ orbit in darkness.

% SLIDE 6: 7D STATE EXPANSION & EQUATIONS OF MOTION
% ==============================================================================
\begin{frame}{Finite Maneuvers: Dynamic Equations and 7D State Vector}
    \begin{columns}[T]
        \begin{column}{0.54\textwidth}
            \begin{block}{Translational Dynamics and Mass Depletion}
                \scriptsize
                Under finite thrust, vehicle mass depletes continuously as propellant is expelled, coupling vehicle acceleration directly to time:
                \begin{equation}
                    \ddot{\mathbf{r}} = -\frac{\mu}{r^3}\mathbf{r} + \mathbf{a}_{\text{perturbations}} + \frac{\mathbf{T}}{m(t)}
                \end{equation}
                \begin{equation}
                    \dot{m}(t) = -\frac{\|\mathbf{T}\|}{I_{\text{sp}} g_0}
                \end{equation}
                \begin{itemize}\setlength{\itemsep}{1.5pt}
                    \item $\mathbf{T}$ is the thrust vector; $g_0 = 9.80665\text{ m/s}^2$.
                    \item Because $m(t)$ is in the denominator, constant thrust magnitude $\|\mathbf{T}\|$ produces continuously increasing acceleration over time.
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{0.44\textwidth}
            \begin{exampleblock}{State-Space Expansion ($\mathbb{R}^6 \to \mathbb{R}^7$)}
                \scriptsize
                In Keplerian dynamics, the state vector is 6D ($\mathbf{r}, \mathbf{v}$). Continuous finite burns require expanding the state vector to include mass as a dynamic coordinate:
                \begin{equation}
                    \mathbf{Y}(t) = \begin{bmatrix} \mathbf{r}(t) \\ \mathbf{v}(t) \\ m(t) \end{bmatrix} = \begin{bmatrix} x \\ y \\ z \\ v_x \\ v_y \\ v_z \\ m \end{bmatrix} \in \mathbb{R}^7
                \end{equation}
                The first-order derivative integrated by numerical solvers is:
                \begin{equation}
                    \frac{d\mathbf{Y}}{dt} = \begin{bmatrix} \mathbf{v} \\ \mathbf{a}_{\text{grav}} + \mathbf{a}_{\text{thrust}} + \mathbf{a}_{\text{pert}} \\ -\|\mathbf{T}\| / (I_{\text{sp}} g_0) \end{bmatrix}
                \end{equation}
            \end{exampleblock}
        \end{column}
    \end{columns}
\end{frame}

                \scriptsize
                \begin{itemize}\setlength{\itemsep}{2pt}\setlength{\topsep}{1pt}

% SLIDE 8: EDELBAUM'S OPTIMAL STEERING
% ==============================================================================
\begin{frame}{Non-Coplanar Transfers: Edelbaum's Optimal Steering}
    \begin{columns}[T]
        \begin{column}{0.53\textwidth}
            \begin{block}{Simultaneous Altitude Raise and Inclination Change}
                \scriptsize
                \begin{itemize}\setlength{\itemsep}{1.2pt}\setlength{\topsep}{0pt}
                    \item Sequential maneuvers (spiraling out, then executing nodal burns) are severely inefficient for electric propulsion.
                    \item \textbf{Edelbaum (1961):} Closed-form analytical $\Delta V$ for simultaneous continuous transfer between circular orbits:
                    \par\vspace{1.5pt}
                    {\centering $\displaystyle \Delta V = \sqrt{v_1^2 + v_2^2 - 2v_1 v_2 \cos\left(\frac{\pi}{2}\Delta i\right)}$ \par}
                    \vspace{1.5pt}
                    \item $v_1, v_2$: Circular velocities; $\Delta i$: plane change in radians.
                \end{itemize}
            \end{block}

            \vspace{-0.22cm}

            \begin{block}{The Continuous Thrust Penalty Factor ($\pi/2$)}
                \scriptsize
                \begin{itemize}\setlength{\itemsep}{1.2pt}\setlength{\topsep}{0pt}
                    \item Impulsive burns change inclination at nodes with $100\%$ geometric efficiency.
                    \item Continuous thrust fires over entire orbits; at $90^\circ$ from nodes, normal thrust produces zero plane change.
                    \item Orbit-averaged efficiency is $2/\pi$, yielding the $\Delta V$ penalty factor $\frac{\pi}{2} \approx 1.57$ inside the cosine argument.
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{0.45\textwidth}
            \begin{alertblock}{The Synergy of Yaw Steering}
                \scriptsize
                \begin{itemize}\setlength{\itemsep}{1.2pt}\setlength{\topsep}{0pt}
                    \item Tilting thrust out-of-plane by yaw angle $\beta$ yields normal acceleration while retaining tangential drive.
                    \item \textbf{Small Angle Power:} Tilting by $\beta = 5^\circ$ yields out-of-plane force while preserving:
                    \par\vspace{1pt}
                    {\centering $\cos(5^\circ) = 0.996 \implies 99.6\%\text{ forward thrust}$ \par}
                \end{itemize}
            \end{alertblock}

\begin{frame}[fragile]{Numerical Trajectory Propagation: Python Derivative Engine}
    \begin{columns}[T]
        \begin{column}{0.49\textwidth}
            \begin{block}{Algorithmic Formulation}
                \scriptsize
                \begin{enumerate}\setlength{\itemsep}{1.5pt}\setlength{\topsep}{1pt}
                    \item Unpack $\mathbf{Y} \in \mathbb{R}^7$: position $\mathbf{r}$, velocity $\mathbf{v}$, and mass $m$.
                    \item Central body gravity acceleration: $\mathbf{a}_{\text{grav}} = -\frac{\mu}{\|\mathbf{r}\|^3}\mathbf{r}$.
                    \item Tangential unit vector: $\hat{\mathbf{u}}_v = \mathbf{v}/\|\mathbf{v}\|$.
                    \item Thrust acceleration: $\mathbf{a}_{\text{thrust}} = \frac{T}{m} \hat{\mathbf{u}}_v$.
                    \item Mass depletion rate: $\dot{m} = -T / (I_{\text{sp}} g_0)$.
                    \item Derivative vector: $d\mathbf{Y}/dt = [\mathbf{v},\, \mathbf{a}_{\text{total}},\, \dot{m}]^T$.
                \end{enumerate}
            \end{block}
            \vspace{-0.1cm}
            \begin{alertblock}{Solver Selection}
                \scriptsize
                Propagated via adaptive Runge-Kutta solvers (e.g., SciPy's \texttt{DOP853} or \texttt{RK45}).
            \end{alertblock}
        \end{column}

        \begin{column}{0.49\textwidth}
\begin{lstlisting}[style=pythoncode]
import numpy as np
from scipy.integrate import solve_ivp

def low_thrust_ode(t, state, mu, thrust, isp, g0):
    # 1. Unpack 7D State Vector
    r_vec = state[0:3]
    v_vec = state[3:6]
    m     = state[6]
    
    # 2. Central Body Gravity Acceleration
    r_mag  = np.linalg.norm(r_vec)
    a_grav = (-mu / r_mag**3) * r_vec
    
    # 3. Tangential Steering Unit Vector
    v_mag    = np.linalg.norm(v_vec)
    u_v      = v_vec / v_mag
    a_thrust = (thrust / m) * u_v
    
    # 4. Total Acceleration & Mass Depletion
    a_total = a_grav + a_thrust
    mdot    = -thrust / (isp * g0)
    
    # 5. Return 7D Derivative Array
    return np.concatenate((v_vec, a_total, [mdot]))
