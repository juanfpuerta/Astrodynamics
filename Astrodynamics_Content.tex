\section{Introduction to Perturbed Orbits}

\begin{frame}{Real World vs. Two-Body Keplerian Motion}
    \begin{block}{Ideal Two-Body Acceleration}
        $$\mathbf{a}_{\text{Kepler}} = -\frac{\mu}{r^3}\mathbf{r}$$
        Assumes point masses, spherical symmetry, and a vacuum environment.
    \end{block}

    \vspace{0.2cm}

    \begin{block}{Perturbed Equation of Motion}
        $$\ddot{\mathbf{r}} = -\frac{\mu}{r^3}\mathbf{r} + \mathbf{a}_{\text{perturbation}}$$
        where the perturbing accelerations are modeled as:
        $$\mathbf{a}_{\text{perturbation}} = \mathbf{a}_{J_2} + \mathbf{a}_{\text{drag}} + \mathbf{a}_{\text{srp}} + \mathbf{a}_{\text{mag}} + \mathbf{a}_{3\text{rd-body}}$$
    \end{block}

    \begin{itemize}
        \item Causes orbital elements $(a, e, i, \Omega, \omega, \nu)$ to deviate from constant Keplerian conics.
        \item Divided into \textbf{gravitational} (conservative) and \textbf{non-gravitational} (dissipative/external) forces.
    \end{itemize}
\end{frame}

\section{Earth Oblateness ($J_2$ Perturbation)}

\begin{frame}{Earth Oblateness ($J_2$): Physics and Potential Expansion}
    \begin{columns}
        \begin{column}{0.55\textwidth}
            \begin{itemize}
                \item Earth's diurnal rotation produces equatorial bulging ($R_e \approx 6378\text{ km}$, $R_p \approx 6356\text{ km}$).
                \item Represented via Spherical Harmonics Expansion of geopotential $V(r, \phi)$:
            \end{itemize}
            \begin{equation*}
                V(r,\phi) = -\frac{\mu}{r}\left[ 1 - \sum_{n=2}^{\infty} J_n \left(\frac{R_e}{r}\right)^n P_n(\sin\phi) \right]
            \end{equation*}
            \begin{itemize}
                \item $J_2 \approx 1.08263 \times 10^{-3}$ is $\sim 1000\times$ larger than higher zonal harmonics ($J_3, J_4$).
            \end{itemize}
        \end{column}

        \begin{column}{0.42\textwidth}
            \begin{block}{$J_2$ Acceleration}
                $$\mathbf{a}_{J_2} = \nabla V_{J_2}$$
                \begin{itemize}
                    \item Creates a non-uniform gravitational torque pulling toward the equator.
                    \item Preserves angular momentum magnitude on average, but rotates the orbital plane in space.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{$J_2$ Secular Effects: Precession Rates}
    $J_2$ produces secular (long-term drift) changes in two primary classical orbital elements:

    \begin{block}{Nodal Precession Rate ($\dot{\Omega}$)}
        $$\dot{\Omega} = -\frac{3}{2} J_2 \left(\frac{R_e}{p}\right)^2 n \cos i \quad \text{[rad/s]}$$
        \begin{itemize}
            \item Precesses westward (retrograde) for prograde orbits ($i < 90^\circ$).
            \item Zero drift for polar orbits ($i = 90^\circ$).
        \end{itemize}
    \end{block}

    \begin{block}{Perigee Precession Rate ($\dot{\omega}$)}
        $$\dot{\omega} = \frac{3}{4} J_2 \left(\frac{R_e}{p}\right)^2 n \left( 5\cos^2 i - 1 \right) \quad \text{[rad/s]}$$
        \begin{itemize}
            \item Drift vanishes at the \textbf{critical inclination}: $i = 63.4^\circ$ or $116.6^\circ$ ($\cos^2 i = 1/5$).
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Engineering Applications of $J_2$}
    \begin{columns}
        \begin{column}{0.45\textwidth}
            \begin{block}{Sun-Synchronous Orbits (SSO)}
                \begin{itemize}
                    \item Target: $\dot{\Omega} = +0.9856^\circ/\text{day}$ ($360^\circ / 365.25\text{ days}$).
                    \item Matches Earth's mean orbital revolution rate around the Sun.
                    \item Ground track experiences constant local solar time illumination (ideal for optical remote sensing).
                    \item Typical LEO altitude: $\sim 600-800\text{ km}$, $i \approx 98^\circ$.
                \end{itemize}
            \end{block}
        \end{column}
        
        \begin{column}{0.45\textwidth}
            \begin{block}{Molniya Orbits}
                \begin{itemize}
                    \item Highly eccentric 12-hour period orbit ($e \approx 0.74$).
                    \item Inclination locked to $i = 63.4^\circ$.
                    \item Freezes argument of perigee ($\dot{\omega} = 0$), preventing perigee from drifting away from the Southern Hemisphere.
                    \item Maximizes dwell time over high northern latitudes.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

%--Slide 1 Expansion
\begin{frame}{The Geopotential Spectrum: Spherical Harmonics}
    The non-spherical mass distribution of Earth is modeled by expanding $V(r, \phi, \lambda)$ in spherical harmonics:

    {\small
    \begin{equation*}
        V(r,\phi,\lambda) = -\frac{\mu}{r} \left[ 1 + \sum_{n=2}^{\infty} \left( \frac{R_e}{r} \right)^n \left( -J_n P_n(\sin\phi) + \sum_{m=1}^{n} P_{nm}(\sin\phi) \left( C_{nm}\cos m\lambda + S_{nm}\sin m\lambda \right) \right) \right]
    \end{equation*}
    }

    \vspace{-0.2cm}

    \begin{columns}[t]
        \begin{column}{0.32\textwidth}
            \begin{block}{\small Zonal Harmonics ($J_n$)}
                \small
                \begin{itemize}
                    \item Order $m = 0$.
                    \item Parallel latitudinal bands.
                    \item Dominant term: $J_2$.
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{0.32\textwidth}
            \begin{block}{\small Tesseral Harmonics}
                \small
                \begin{itemize}
                    \item $n \neq m \neq 0$.
                    \item Lat. \& Long. dependent.
                    \item 2D checkerboard grid.
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{0.32\textwidth}
            \begin{block}{\small Sectoral Harmonics}
                \small
                \begin{itemize}
                    \item Order $m = n$.
                    \item Longitude-dependent.
                    \item "Orange-slice" wedges.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

% Slide 3: Legendre Polynomials and J2 Potential
% ------------------------------------------------------------------------------
\begin{frame}{The $J_2$ Term: Legendre Polynomial Expansion}
    \begin{columns}[t]
        \begin{column}{0.48\textwidth}
            \begin{block}{Equatorial Bulge \& $P_2$}
                \begin{itemize}
                    \item $n=2$ zonal harmonic models Earth's \textbf{equatorial bulge} ($R_e \approx 6378\text{ km}$, $R_p \approx 6356\text{ km}$).
                    \item $2^{\text{nd}}$-degree \textbf{Legendre Polynomial}:
                    \begin{equation*}
                        P_2(\sin\phi) = \frac{1}{2}\left( 3\sin^2\phi - 1 \right)
                    \end{equation*}
                    where $\phi$ is geocentric latitude ($\sin\phi = z/r$).
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{0.48\textwidth}
            \begin{block}{$J_2$ Perturbing Potential ($V_{J_2}$)}
                \begin{equation*}
                    V_{J_2}(r, \phi) = \frac{\mu R_e^2 J_2}{2 r^3} \left( 3\sin^2\phi - 1 \right)
                \end{equation*}
                where $J_2 \approx 1.08263 \times 10^{-3}$ (dimensionless).
            \end{block}

            \begin{alertblock}{Key Observation}
                Since $J_2 \sim 10^{-3}$ while $J_3, J_4 \sim 10^{-6}$, $J_2$ accounts for over \textbf{99.9\%} of all non-spherical gravitational perturbations!
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

% ------------------------------------------------------------------------------
% Slide 4: Deriving Perturbing Acceleration Vector
% ------------------------------------------------------------------------------
\begin{frame}{Deriving the $J_2$ Perturbing Acceleration Vector}
    Taking the gradient of the potential ($\mathbf{a}_{J_2} = \nabla V_{J_2}$) in Geocentric Equatorial frame ($x, y, z$):

    \begin{equation*}
        \mathbf{a}_{J_2} = -\frac{3 J_2 \mu R_e^2}{2 r^5} 
        \begin{bmatrix}
            x \left( 1 - 5\frac{z^2}{r^2} \right) \\
            y \left( 1 - 5\frac{z^2}{r^2} \right) \\
            z \left( 3 - 5\frac{z^2}{r^2} \right)
        \end{bmatrix}
    \end{equation*}

    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Physical Mechanism}
                \begin{itemize}
                    \item $\mathbf{a}_{J_2}$ creates a non-central force directed toward the equatorial plane.
                    \item Exerts an out-of-plane torque on tilted orbits ($i \neq 0^\circ, 90^\circ$).
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Conservation Laws}
                \begin{itemize}
                    \item Axisymmetric about Z-axis $\implies$ $h_z$ is conserved!
                    \item Conservative force $\implies$ Specific mechanical energy $E$ is conserved on average.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

% ------------------------------------------------------------------------------
% Slide 5: The Period-Averaging Process
% ------------------------------------------------------------------------------
\begin{frame}{The Orbit-Averaging Process (Lagrange Planetary Equations)}
    To separate slow, cumulative long-term trends (secular variations) from fast intra-orbit oscillations (short-period variations), we integrate Lagrange's Planetary Equations over one full orbital period $T$:

    \begin{equation*}
        \bar{\dot{\Omega}} = \frac{1}{2\pi} \int_{0}^{2\pi} \dot{\Omega}(\nu) \, d\nu, \qquad 
        \bar{\dot{\omega}} = \frac{1}{2\pi} \int_{0}^{2\pi} \dot{\omega}(\nu) \, d\nu
    \end{equation*}

    \begin{block}{First-Order Secular Drift Formulas}
        \begin{align*}
            \dot{\Omega} &= -\frac{3}{2} J_2 \left(\frac{R_e}{p}\right)^2 n \cos i \quad \text{[rad/s]} \\
            \dot{\omega} &= \frac{3}{4} J_2 \left(\frac{R_e}{p}\right)^2 n \left( 5\cos^2 i - 1 \right) \quad \text{[rad/s]}
        \end{align*}
        where $p = a(1-e^2)$ is the semi-latus rectum, and $n = \sqrt{\mu/a^3}$ is the mean motion.
    \end{block}
\end{frame}

\begin{frame}{Worked Example 1: Sun-Synchronous Orbit (SSO) Design}
    \begin{exampleblock}{Problem Statement}
        A remote-sensing satellite is launched into a circular orbit at an altitude of $h = 800\text{ km}$. Calculate the required orbital inclination $i$ so that the orbit is Sun-Synchronous ($\dot{\Omega}_{\text{SSO}} = +0.9856^\circ/\text{day}$).
    \end{exampleblock}

    \textbf{Given Constants:}
    \begin{itemize}
        \item $R_e = 6378.137\text{ km}$, $\mu = 398600.44\text{ km}^3/\text{s}^2$, $J_2 = 1.08263 \times 10^{-3}$
        \item Solar precession rate: $\dot{\Omega} = 0.9856^\circ/\text{day} = 1.99106 \times 10^{-7}\text{ rad/s}$
    \end{itemize}

    \textbf{Step 1: Compute Orbital Geometry \& Mean Motion}
    \begin{align*}
        a &= R_e + h = 6378.137 + 800 = 7178.137\text{ km} \quad (\text{and } p = a \text{ since } e=0) \\
        n &= \sqrt{\frac{\mu}{a^3}} = \sqrt{\frac{398600.44}{(7178.137)^3}} = 1.0390 \times 10^{-3}\text{ rad/s}
    \end{align*}
\end{frame}

% ------------------------------------------------------------------------------
% Slide 7: Worked Example 1 (Solution Continued)
% ------------------------------------------------------------------------------
\begin{frame}{Worked Example 1: Solution Continued}
    \begin{columns}[t]
        \begin{column}{0.55\textwidth}
            \textbf{Step 2: Substitute into Nodal Drift Formula}
            {\small
            \begin{align*}
                \dot{\Omega} &= -\frac{3}{2} J_2 \left(\frac{R_e}{a}\right)^2 n \cos i \\[3pt]
                1.99106 \times 10^{-7} &= -\frac{3}{2} (1.08263 \times 10^{-3}) \left(\frac{6378.137}{7178.137}\right)^2 \\
                &\quad \times (1.0390 \times 10^{-3}) \cos i \\[3pt]
                1.99106 \times 10^{-7} &= -1.33235 \times 10^{-6} \cos i
            \end{align*}
            }

            \textbf{Step 3: Solve for Inclination $i$}
            {\small
            \begin{align*}
                \cos i &= \frac{1.99106 \times 10^{-7}}{-1.33235 \times 10^{-6}} = -0.14944 \\[3pt]
                i &= \arccos(-0.14944) = \mathbf{98.59^\circ}
            \end{align*}
            }
        \end{column}

        \begin{column}{0.41\textwidth}
            \begin{block}{Result Analysis}
                The orbit must be \textbf{retrograde} ($i > 90^\circ$) so that $\cos i < 0$, yielding a positive (eastward) nodal precession rate ($\dot{\Omega} > 0$).
            \end{block}

            \begin{itemize}
                \item \small Precession matches Earth's mean revolution around the Sun ($0.9856^\circ/\text{day}$).
                \item \small Guarantees identical solar illumination on every pass.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

% Slide 8: Worked Example 2 (Molniya & Critical Inclination)
% ------------------------------------------------------------------------------
\begin{frame}{Worked Example 2: Molniya Orbit Drift Rates}
    \begin{exampleblock}{Problem Statement}
        A communications satellite flies in a 12-hour Molniya orbit with $a = 26561\text{ km}$ and $e = 0.74$. 
        \begin{enumerate}
            \item Show why setting $i = 63.435^\circ$ freezes the argument of perigee ($\dot{\omega} = 0$).
            \item Calculate the secular nodal precession rate $\dot{\Omega}$ (in deg/day) at this inclination.
        \end{enumerate}
    \end{exampleblock}

    \textbf{Solution Part 1: Critical Inclination Condition}
    $$\dot{\omega} = \frac{3}{4} J_2 \left(\frac{R_e}{p}\right)^2 n \left( 5\cos^2 i - 1 \right) = 0 \implies 5\cos^2 i - 1 = 0$$
    $$\cos i = \frac{1}{\sqrt{5}} \implies i = \arccos(0.44721) = \mathbf{63.435^\circ}$$
    This prevents the perigee from moving away from the Southern Hemisphere!
\end{frame}

% ------------------------------------------------------------------------------
% Slide 9: Worked Example 2 (Solution Part 2)
% ------------------------------------------------------------------------------
\begin{frame}{Worked Example 2: Solution Part 2}
    \textbf{Solution Part 2: Nodal Precession Calculation}
    \begin{align*}
        p &= a(1 - e^2) = 26561 \times (1 - 0.74^2) = 12016.3\text{ km} \\
        n &= \sqrt{\frac{398600.44}{(26561)^3}} = 1.4584 \times 10^{-4}\text{ rad/s}
    \end{align*}

    Substitute into $\dot{\Omega}$:
    $$\dot{\Omega} = -\frac{3}{2} (1.08263 \times 10^{-3}) \left(\frac{6378.137}{12016.3}\right)^2 (1.4584 \times 10^{-4}) \cos(63.435^\circ)$$
    $$\dot{\Omega} = -3.003 \times 10^{-8}\text{ rad/s}$$

    Convert to degrees per day:
    $$\dot{\Omega}_{\text{daily}} = (-3.003 \times 10^{-8}\text{ rad/s}) \times \left(\frac{180}{\pi}\right) \times 86400\text{ s/day} = \mathbf{-0.148^\circ/\text{day}}$$
\end{frame}

\section{Atmospheric Drag}

\begin{frame}{Atmospheric Drag: Physical Principles}
    \begin{block}{Drag Acceleration Model}
        $$\mathbf{a}_{\text{drag}} = -\frac{1}{2} \rho \left(\frac{C_D A}{m}\right) v_{\text{rel}} \mathbf{v}_{\text{rel}}$$
    \end{block}

    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Key Parameters:}
            \begin{itemize}
                \item $\rho$: Local atmospheric density (exponential falloff with altitude).
                \item $C_D$: Drag coefficient ($\sim 2.0 - 2.2$ for complex geometries).
                \item $A/m$: Cross-sectional area-to-mass ratio [$\text{m}^2/\text{kg}$].
                \item $\mathbf{v}_{\text{rel}}$: Velocity relative to co-rotating atmosphere ($\mathbf{v} - \boldsymbol{\omega}_e \times \mathbf{r}$).
            \end{itemize}
        \end{column}

        \begin{column}{0.5\textwidth}
            \begin{block}{Ballistic Coefficient ($B^*$)}
                $$B^* = \frac{C_D A}{m}$$
                Measures vehicle susceptibility to drag. Spacecraft with large deployable arrays exhibit higher decay rates.
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Effects of Drag on Orbital Propagation}
    \begin{columns}
        \begin{column}{0.55\textwidth}
            \begin{itemize}
                \item \textbf{Energy Loss:} Dissipates total orbital mechanical energy ($E = -\mu / 2a$).
                \item \textbf{Semi-Major Axis Decay:}
                $$\frac{da}{dt} \approx -2 a^2 \frac{a_{\text{drag}}}{v} < 0$$
                \item \textbf{Orbital Circularization:} Peak drag acts at perigee (highest atmospheric density), lowering apogee much faster than perigee, driving $e \to 0$.
                \item \textbf{Paradoxical Effect:} As $a$ drops, orbital speed $v = \sqrt{\mu/a}$ actually increases!
            \end{itemize}
        \end{column}
        
        \begin{column}{0.42\textwidth}
            \begin{block}{Density Variability Drivers}
                Upper thermospheric density $\rho(h)$ varies dynamically due to:
                \begin{itemize}
                    \item 11-year Solar Activity Cycle (Extreme UV solar heating).
                    \item Geomagnetic storms.
                    \item Day/Night diurnal atmospheric bulge.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\section{Upper Atmospheric Modeling}

\begin{frame}{Exponential Atmosphere \& Atmospheric Rotation}
    \begin{columns}[t]
        \begin{column}{0.48\textwidth}
            \begin{block}{1. Exponential Density Model}
                Upper thermospheric density falls off exponentially with geometric altitude $h$:
                \begin{equation*}
                    \rho(h) = \rho_0 \exp\left(-\frac{h - h_0}{H}\right)
                \end{equation*}
                where $H$ is the \textbf{atmospheric scale height}:
                \begin{equation*}
                    H = \frac{k_B T_{\text{atm}}}{\bar{M} g}
                \end{equation*}
                ($H \approx 50\text{--}80\text{ km}$ in LEO, varying dynamically with solar EUV heating).
            \end{block}
        \end{column}

        \begin{column}{0.48\textwidth}
            \begin{block}{2. Co-Rotating Atmosphere Velocity}
                The atmosphere rotates alongside Earth's spin $\boldsymbol{\omega}_e = [0, 0, \omega_e]^T$. Relative velocity vector $\mathbf{v}_{\text{rel}}$ is:
                \begin{equation*}
                    \mathbf{v}_{\text{rel}} = \mathbf{v} - \boldsymbol{\omega}_e \times \mathbf{r} = \begin{bmatrix} v_x + \omega_e y \\ v_y - \omega_e x \\ v_z \end{bmatrix}
                \end{equation*}
                Atmospheric rotation reduces relative drag velocity for prograde orbits ($i < 90^\circ$).
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\section{Analytical Derivations of Orbital Decay}

\begin{frame}{Derivation: Semi-Major Axis Decay Rate ($\frac{da}{dt}$)}
    \begin{columns}[t]
        \begin{column}{0.52\textwidth}
            Specific orbital energy: $E = -\frac{\mu}{2a}$.
            
            \smallskip
            \textbf{Step 1: Energy Rate Derivative}
            \begin{equation*}
                \frac{dE}{dt} = \frac{\mu}{2a^2} \frac{da}{dt}
            \end{equation*}

            \textbf{Step 2: Work Done by Drag}
            \begin{equation*}
                \frac{dE}{dt} = \mathbf{a}_{\text{drag}} \cdot \mathbf{v} = -a_{\text{drag}} v
            \end{equation*}

            \textbf{Step 3: Equating Rates ($v = \sqrt{\mu/a}$)}
            \begin{equation*}
                \frac{\mu}{2a^2} \frac{da}{dt} = -a_{\text{drag}} \sqrt{\frac{\mu}{a}} \implies \frac{da}{dt} = -\frac{2 a_{\text{drag}}}{n}
            \end{equation*}
        \end{column}

        \begin{column}{0.44\textwidth}
            \begin{block}{Fundamental Decay Formula}
                For circular orbits ($n = \sqrt{\mu/a^3}$):
                \begin{equation*}
                    \frac{da}{dt} = -\frac{2 a_{\text{drag}}}{n}
                \end{equation*}
                Substituting $a_{\text{drag}} = \frac{1}{2}\rho v^2 \left(\frac{C_D A}{m}\right)$:
                \begin{equation*}
                    \frac{da}{dt} = -\rho \left(\frac{C_D A}{m}\right) \sqrt{\mu a} \quad \text{[m/s]}
                \end{equation*}
            \end{block}

            \begin{itemize}
                \item \small Rate is strictly negative ($\frac{da}{dt} < 0$).
                \item \small Scales linearly with density $\rho$ and area-to-mass ratio $A/m$.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Derivation: Orbital Period Decay Rate ($\frac{dT}{dt}$)}
    Using Kepler's $3^{\text{rd}}$ Law relating Period $T$ and Semi-Major Axis $a$:
    \begin{equation*}
        T = 2\pi \sqrt{\frac{a^3}{\mu}} \implies \frac{dT}{dt} = \frac{3\pi}{v} \frac{da}{dt}
    \end{equation*}

    Substitute $\frac{da}{dt} = -\rho \left(\frac{C_D A}{m}\right) v^2 a$:
    \begin{equation*}
        \frac{dT}{dt} = \frac{3\pi}{v} \left[ -\rho \left(\frac{C_D A}{m}\right) v^2 a \right] = -3\pi a \rho \left(\frac{C_D A}{m}\right) v
    \end{equation*}

    \begin{alertblock}{King-Hele's Orbital Period Change Relation}
        \begin{equation*}
            \frac{dT}{dt} = -3\pi a \rho \left(\frac{C_D A}{m}\right) \quad \text{[dimensionless, s/s]}
        \end{equation*}
        Directly links orbital period drop per orbit to the ballistic parameter $B^* = \frac{C_D A}{m}$.
    \end{alertblock}
\end{frame}

\begin{frame}{Asymmetric Drag in Elliptical Orbits \& Circularization}
    In an elliptical orbit ($e > 0$), density varies exponentially between periapsis ($r_p$) and apoapsis ($r_a$):
    \begin{equation*}
        \frac{\rho_{\text{periapsis}}}{\rho_{\text{apoapsis}}} = \exp\left( \frac{r_a - r_p}{H} \right) = \exp\left( \frac{2 a e}{H} \right)
    \end{equation*}

    \begin{columns}[t]
        \begin{column}{0.48\textwidth}
            \begin{block}{Physical Behavior}
                \begin{itemize}
                    \item Peak drag impulses occur localized around \textbf{periapsis}.
                    \item Drag at periapsis removes kinetic energy, selectively \textbf{lowering apoapsis altitude} ($r_a \downarrow$).
                    \item Periapsis altitude remains nearly constant ($r_p \approx \text{const}$).
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{0.48\textwidth}
            \begin{block}{Secular Eccentricity Decay}
                \begin{equation*}
                    \frac{de}{dt} \approx -\rho_p \left(\frac{C_D A}{m}\right) \sqrt{\frac{\mu}{a}} I_1\left(\frac{ae}{H}\right) e^{-\frac{ae}{H}}
                \end{equation*}
                \begin{itemize}
                    \item Forces $e \to 0$ (orbital circularization) prior to final atmospheric plunge.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Orbital Lifetime Estimation in Circular Orbit}
    Assuming constant scale height $H$, the remaining lifetime $L_{\text{lifetime}}$ of a satellite decaying from initial altitude $h_0$ can be approximated analytically by integrating the scale height:

    \begin{block}{Linearized Lifetime Formula}
        \begin{equation*}
            L_{\text{lifetime}} \approx \frac{H}{|\dot{a}_0|} = \frac{H}{\rho_0 \left(\frac{C_D A}{m}\right) \sqrt{\mu a_0}}
        \end{equation*}
        where $|\dot{a}_0|$ is the initial daily altitude loss rate at $h_0$.
    \end{block}

    \medskip

    \begin{exampleblock}{Key Engineering Insights}
        \begin{itemize}
            \item \textbf{The "Plunge" Effect:} Because density increases exponentially as altitude drops, $80\%$ of a satellite's total operational lifetime is spent in the first $100\text{ km}$ of decay!
            \item Once altitude drops below $\sim 180\text{--}200\text{ km}$, final re-entry occurs within hours.
        \end{itemize}
    \end{exampleblock}
\end{frame}

\section{Step-by-Step Worked Examples}

\begin{frame}{Worked Example 1: Space Station Altitude \& Period Loss}
    \begin{exampleblock}{Problem Statement}
        A small space station module ($m = 2000\text{ kg}$) with an effective cross-sectional area $A = 15\text{ m}^2$ flies in a circular LEO at $h = 350\text{ km}$. Given $C_D = 2.2$ and thermospheric density $\rho = 3.8 \times 10^{-12}\text{ kg/m}^3$:
        \begin{enumerate}
            \item Calculate the drag acceleration $a_{\text{drag}}$.
            \item Calculate the daily altitude decay rate $\Delta a_{\text{daily}}$ (in km/day).
            \item Calculate the period drop per day $\Delta T_{\text{daily}}$ (in seconds/day).
        \end{enumerate}
    \end{exampleblock}

    \textbf{Given Constants:}
    \begin{itemize}
        \item $R_e = 6378.137\text{ km}$, $\mu = 398600.44\text{ km}^3/\text{s}^2$.
    \end{itemize}
\end{frame}

\begin{frame}{Worked Example 1: Detailed Analytical Solution}
    \textbf{Step 1: Compute Orbit Geometry \& Velocity}
    \begin{align*}
        a &= R_e + h = 6378.137 + 350 = 6728.137\text{ km} = 6.728137 \times 10^6\text{ m} \\
        v &= \sqrt{\frac{\mu}{a}} = \sqrt{\frac{3.9860044 \times 10^{14}}{6.728137 \times 10^6}} = 7697.16\text{ m/s}
    \end{align*}

    \textbf{Step 2: Calculate Drag Acceleration $a_{\text{drag}}$}
    \begin{align*}
        a_{\text{drag}} &= \frac{1}{2} \rho \left(\frac{C_D A}{m}\right) v^2 = \frac{1}{2} (3.8 \times 10^{-12}) \left(\frac{2.2 \times 15}{2000}\right) (7697.16)^2 \\
        &= \mathbf{1.857 \times 10^{-6}\text{ m/s}^2}
    \end{align*}

    \textbf{Step 3: Daily Altitude Decay Rate $\Delta a_{\text{daily}}$}
    \begin{align*}
        n &= \sqrt{\frac{\mu}{a^3}} = 1.1438 \times 10^{-3}\text{ rad/s} \implies \frac{da}{dt} = -\frac{2 a_{\text{drag}}}{n} = -3.247 \times 10^{-3}\text{ m/s} \\
        \Delta a_{\text{daily}} &= (-3.247 \times 10^{-3}\text{ m/s}) \times 86400\text{ s/day} = \mathbf{-0.2805\text{ km/day}}
    \end{align*}
\end{frame}

\begin{frame}{Worked Example 1: Solution Continued}
    \textbf{Step 4: Daily Orbital Period Drop $\Delta T_{\text{daily}}$}
    \par\smallskip
    Using the King-Hele period drop formula:
    \begin{align*}
        \frac{dT}{dt} &= -3\pi a \rho \left(\frac{C_D A}{m}\right) \\
        &= -3\pi (6.728137 \times 10^6) (3.8 \times 10^{-12}) (0.0165) = -3.974 \times 10^{-6}\text{ s/s}
    \end{align*}

    Convert rate into total period change per 24-hour day:
    \begin{equation*}
        \Delta T_{\text{daily}} = \left( \frac{dT}{dt} \right) \times 86400 = (-3.974 \times 10^{-6}) \times 86400 = \mathbf{-0.343\text{ seconds/day}}
    \end{equation*}

    \begin{block}{Operational Significance}
        Loss of $0.343\text{ seconds/day}$ in period accumulates a \textbf{downrange along-track position error} of $\Delta x = v \cdot \Delta T \approx 2.6\text{ km}$ every single day if drag is unmodeled!
    \end{block}
\end{frame}

\section{Solar Radiation Pressure (SRP)}

\begin{frame}{Solar Radiation Pressure (SRP): Physics}
    Photons emitted by the Sun exert momentum upon impact with satellite surfaces.

    \begin{block}{SRP Acceleration Model ("Cannonball Model")}
        $$\mathbf{a}_{\text{srp}} = - P_{\odot} C_r \left(\frac{A}{m}\right) \left(\frac{1 \text{ AU}}{r_{\odot}}\right)^2 \hat{\mathbf{u}}_{\odot}$$
    \end{block}

    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Parameters:}
            \begin{itemize}
                \item $P_{\odot}$: Solar radiation constant at $1\text{ AU} \approx 4.56 \times 10^{-6}\text{ N/m}^2$.
                \item $C_r$: Radiation reflectivity ($1.0$ pure absorption, $2.0$ total specular reflection).
                \item $\hat{\mathbf{u}}_{\odot}$: Unit vector pointing toward Sun.
            \end{itemize}
        \end{column}
        
        \begin{column}{0.5\textwidth}
            \begin{block}{Eclipse / Shadow Function}
                $$\nu_{\text{shadow}} = \begin{cases} 
                    1, & \text{Direct Sunlight} \\
                    (0, 1), & \text{Penumbra} \\
                    0, & \text{Umbra (Total Shadow)}
                \end{cases}$$
                Requires tracking Earth's shadow cone geometries during orbit propagation.
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{SRP Effects at High Altitudes and Solar Sailing}
    \begin{itemize}
        \item \textbf{Dominance in HEO/GEO:} While Drag dominates low LEO ($< 600\text{ km}$), SRP becomes the leading non-gravitational force in MEO, GEO, and high-altitude orbits.
        \item \textbf{Long-term Orbital Drift:} Induces periodic long-term variations in eccentricity ($e$) and inclination ($i$).
        \item \textbf{Space Debris Vulnerability:} Objects with very high $A/m$ ratios (e.g., thermal insulation fragments) experience massive SRP pertubation, leading to rapid eccentricity pumping and premature atmospheric entry.
    \end{itemize}

    \begin{exampleblock}{Propulsive Solar Sailing}
        Solar sails intentionally maximize $A/m$ ratios ($> 10\text{ m}^2/\text{kg}$) to harness $\mathbf{a}_{\text{srp}}$ for propellantless interplanetary maneuvering and orbital inclination changes.
    \end{exampleblock}
\end{frame}

\section{Geomagnetic Field Interactions}

\begin{frame}{Geomagnetic Field Interaction Mechanisms}
    Earth's geomagnetic dipole field $\mathbf{B}$ interacts with spacecraft through two primary physical pathways:

    \begin{block}{1. Magnetic Disturbance Torques (Attitude Dynamics)}
        $$\boldsymbol{\tau}_{\text{mag}} = \mathbf{m}_{\text{sat}} \times \mathbf{B}$$
        \begin{itemize}
            \item $\mathbf{m}_{\text{sat}}$: Residual magnetic dipole moment of internal electronics and current loops.
            \item Primary attitude perturbation torque in LEO.
            \item Exploited purposefully using **Magnetorquers** ($\mathbf{m}_{\text{cmd}}$) for attitude control and reaction wheel desaturation.
        \end{itemize}
    \end{block}

    \begin{block}{2. Lorentz Force Acceleration (Translational Dynamics)}
        $$\mathbf{a}_{\text{Lorentz}} = \frac{q_{\text{sat}}}{m} \left( \mathbf{v}_{\text{rel}} \times \mathbf{B} \right)$$
        \begin{itemize}
            \item $q_{\text{sat}}$: Electrostatic charge acquired by spacecraft in ionospheric plasma.
            \item Utilized by **Electrodynamic Tethers (EDTs)** ($I \mathbf{L} \times \mathbf{B}$) to generate drag or boost forces without propellant consumption.
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Summary for Orbit Determination and Estimation}
    \begin{enumerate}
        \item \textbf{$J_2$ Oblateness:} Conservative force causing nodal precession ($\dot{\Omega}$) and perigee rotation ($\dot{\omega}$). Basis of SSO and Molniya orbits.
        \item \textbf{Atmospheric Drag:} Non-conservative force driving semi-major axis decay ($a\downarrow$) and circularization ($e\downarrow$). Dominant perturbation in LEO.
        \item \textbf{Solar Radiation Pressure:} Non-conservative force dominating in GEO/HEO. Induces long-term eccentricity oscillations.
        \item \textbf{Geomagnetic Interactions:} Produces primary disturbance torque for attitude control; can be leveraged via magnetorquers or electrodynamic tethers.
        \item \textbf{Estimation Implications:} Precise Orbit Determination (POD) filters (EKF/Batch) must estimate dynamic atmospheric scaling factors ($B^*$) and solar reflectivity coefficients ($C_r$) alongside spacecraft position and velocity states.
    \end{enumerate}
\end{frame}
